-- Kiểm tra và tạo Database trong SQL Server
IF NOT EXISTS (SELECT * FROM sys.databases WHERE name = 'SE_Career_Roadmap')
BEGIN
    CREATE DATABASE SE_Career_Roadmap;
END
GO

USE SE_Career_Roadmap;
GO

-- ==========================================
-- 1. MODULE: QUẢN LÝ NGƯỜI DÙNG
-- ==========================================
CREATE TABLE Users (
    user_id VARCHAR(50) PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255),
    full_name NVARCHAR(255) NOT NULL, -- Dùng NVARCHAR để lưu tiếng Việt
    role VARCHAR(20) DEFAULT 'STUDENT' CHECK (role IN ('STUDENT', 'STAFF', 'MENTOR')),
    oauth_provider VARCHAR(20) DEFAULT 'LOCAL' CHECK (oauth_provider IN ('GOOGLE', 'LOCAL')),
    oauth_id VARCHAR(255),
    created_at DATETIME DEFAULT GETDATE(),
    gemini_api_key VARCHAR(255) NULL
);

-- ==========================================
-- 2. MODULE: HỒ SƠ HỌC THUẬT & KỸ NĂNG CƠ BẢN
-- ==========================================
CREATE TABLE Academic_Records (
    record_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    course_code VARCHAR(50) NOT NULL,
    course_name NVARCHAR(255) NOT NULL,
    gpa DECIMAL(4,2) NOT NULL,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

CREATE TABLE Skills (
    skill_id VARCHAR(50) PRIMARY KEY,
    skill_name VARCHAR(100) NOT NULL UNIQUE,
    category NVARCHAR(100)
);

CREATE TABLE Student_Skills (
    student_skill_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    skill_id VARCHAR(50) NOT NULL,
    source VARCHAR(30) DEFAULT 'MANUAL' CHECK (source IN ('MANUAL', 'GITHUB_EXTRACTED')),
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (skill_id) REFERENCES Skills(skill_id) ON DELETE CASCADE
);

-- ==========================================
-- 3. MODULE: LỘ TRÌNH HỌC TẬP (DYNAMIC ROADMAP)
-- ==========================================
CREATE TABLE Career_Roles (
    role_id VARCHAR(50) PRIMARY KEY,
    role_name NVARCHAR(150) NOT NULL,
    description NVARCHAR(MAX)
);

CREATE TABLE Role_Prerequisites (
    id VARCHAR(50) PRIMARY KEY,
    role_id VARCHAR(50) NOT NULL,
    skill_id VARCHAR(50) NOT NULL,
    FOREIGN KEY (role_id) REFERENCES Career_Roles(role_id) ON DELETE CASCADE,
    FOREIGN KEY (skill_id) REFERENCES Skills(skill_id) ON DELETE CASCADE
);

CREATE TABLE Roadmaps (
    roadmap_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    target_role_id VARCHAR(50) NOT NULL,
    progress_percent DECIMAL(5,2) DEFAULT 0.00,
    created_at DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (target_role_id) REFERENCES Career_Roles(role_id) ON DELETE CASCADE
);

CREATE TABLE Skill_Nodes (
    node_id VARCHAR(50) PRIMARY KEY,
    roadmap_id VARCHAR(50) NOT NULL,
    skill_id VARCHAR(50) NOT NULL,
    parent_node_id VARCHAR(50),
    status VARCHAR(20) DEFAULT 'PENDING' CHECK (status IN ('PENDING', 'COMPLETED')),
    FOREIGN KEY (roadmap_id) REFERENCES Roadmaps(roadmap_id) ON DELETE CASCADE,
    FOREIGN KEY (skill_id) REFERENCES Skills(skill_id) ON DELETE CASCADE,
    -- SQL Server không cho phép CASCADE trên bảng tự tham chiếu (self-referencing), nên để mặc định (NO ACTION)
    FOREIGN KEY (parent_node_id) REFERENCES Skill_Nodes(node_id) 
);

CREATE TABLE Learning_Resources (
    resource_id VARCHAR(50) PRIMARY KEY,
    skill_id VARCHAR(50) NOT NULL,
    course_id VARCHAR(50) NOT NULL,
    title NVARCHAR(255) NOT NULL,
    url VARCHAR(500) NOT NULL,
    FOREIGN KEY (skill_id) REFERENCES Skills(skill_id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES Courses(course_id) ON DELETE CASCADE
);

-- ==========================================
-- 4. MODULE: HỒ SƠ NĂNG LỰC (E-PORTFOLIO)
-- ==========================================
CREATE TABLE Github_Profiles (
    profile_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL UNIQUE,
    github_username VARCHAR(100) NOT NULL,
    portfolio_url VARCHAR(255) UNIQUE,
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

CREATE TABLE Repositories (
    repo_id VARCHAR(50) PRIMARY KEY,
    profile_id VARCHAR(50) NOT NULL,
    repo_name VARCHAR(255) NOT NULL,
    repo_url VARCHAR(500) NOT NULL,
    ai_summary NVARCHAR(MAX),
    tech_stack NVARCHAR(MAX),
    FOREIGN KEY (profile_id) REFERENCES Github_Profiles(profile_id) ON DELETE CASCADE
);

-- ==========================================
-- 5. MODULE: PHÂN TÍCH THỊ TRƯỜNG (MARKET PULSE)
-- ==========================================
CREATE TABLE Job_Trends (
    trend_id VARCHAR(50) PRIMARY KEY,
    skill_id VARCHAR(50) NOT NULL,
    keyword VARCHAR(100) NOT NULL,
    frequency_count INT DEFAULT 0,
    source_platform VARCHAR(100),
    scrape_date DATE NOT NULL,
    FOREIGN KEY (skill_id) REFERENCES Skills(skill_id) ON DELETE CASCADE
);

-- ==========================================
-- 6. MODULE: TRỢ LÝ ẢO AI (AI MENTOR)
-- ==========================================
CREATE TABLE Mentor_Sessions (
    session_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    created_at DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE
);

CREATE TABLE Chat_Messages (
    message_id VARCHAR(50) PRIMARY KEY,
    session_id VARCHAR(50) NOT NULL,
    sender VARCHAR(10) NOT NULL CHECK (sender IN ('USER', 'AI')),
    content NVARCHAR(MAX) NOT NULL,
    timestamp DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (session_id) REFERENCES Mentor_Sessions(session_id) ON DELETE CASCADE
);


USE SE_Career_Roadmap;
GO

-- ==========================================================
-- 1. TẠO BẢNG QUẢN LÝ MÔN HỌC & CHUẨN ĐẦU RA (MAINFLOW 1)
-- ==========================================================

-- Tạo bảng Courses để lưu trữ thông tin chuẩn hóa của các môn học
CREATE TABLE Courses (
    course_id VARCHAR(50) PRIMARY KEY,
    course_code VARCHAR(50) NOT NULL UNIQUE,
    course_name NVARCHAR(255) NOT NULL,
    credits INT DEFAULT 3,
    total_study_hours INT DEFAULT 0 -- Phục vụ tính toán deadline (Mainflow 2)
);
GO

-- Tạo bảng Course_Learning_Outcomes (Môn này học xong được skill gì)
CREATE TABLE Course_Learning_Outcomes (
    id VARCHAR(50) PRIMARY KEY,
    course_id VARCHAR(50) NOT NULL,
    skill_id VARCHAR(50) NOT NULL,
    outcome_description NVARCHAR(500),
    FOREIGN KEY (course_id) REFERENCES Courses(course_id) ON DELETE CASCADE,
    FOREIGN KEY (skill_id) REFERENCES Skills(skill_id) ON DELETE CASCADE
);
GO

-- ==========================================================
-- 2. CẬP NHẬT LẠI BẢNG ĐIỂM (ACADEMIC RECORDS)
-- Vì bảng cũ lưu tên/mã môn dạng text thuần, ta cần xóa đi và
-- tạo lại để liên kết chuẩn bằng khóa ngoại (course_id) và thêm số lần thi.
-- ==========================================================

-- Xóa bảng cũ (Lưu ý: Nếu có data nháp, data này sẽ bị xóa)
IF OBJECT_ID('Academic_Records', 'U') IS NOT NULL
    DROP TABLE Academic_Records;
GO

-- Tạo lại bảng Academic_Records chuẩn chỉnh hơn
CREATE TABLE Academic_Records (
    record_id VARCHAR(50) PRIMARY KEY,
    user_id VARCHAR(50) NOT NULL,
    course_id VARCHAR(50) NOT NULL, -- Link trực tiếp đến bảng Courses
    gpa DECIMAL(4,2) NOT NULL,
    exam_attempts INT DEFAULT 1,    -- Bổ sung số lần thi (Mainflow 1)
    FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES Courses(course_id) ON DELETE CASCADE
);
GO

-- ==========================================================
-- 3. CẬP NHẬT BẢNG LỘ TRÌNH & ĐIỂM NÚT (MAINFLOW 2)
-- ==========================================================

-- Thêm "Số giờ học cam kết mỗi ngày" vào Lộ trình
ALTER TABLE Roadmaps 
ADD daily_study_hours DECIMAL(4,2) DEFAULT 2.00;
GO

-- Thêm "Deadline" và link "Môn học" vào từng Node trên Lộ trình
ALTER TABLE Skill_Nodes 
ADD deadline DATE,
    course_id VARCHAR(50) NULL;
GO

-- Cài đặt khóa ngoại cho course_id trong Skill_Nodes
ALTER TABLE Skill_Nodes
ADD CONSTRAINT FK_SkillNodes_Courses 
FOREIGN KEY (course_id) REFERENCES Courses(course_id) ON DELETE SET NULL;
GO
